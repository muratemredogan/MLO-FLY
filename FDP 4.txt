import numpy as np 
 
def bucketize_delay(delay_minutes): 
    """Convert continuous delay to categorical class""" 
    if delay_minutes < 15: 
        return 0  # On Time 
    elif delay_minutes < 60: 
        return 1  # Minor Delay 
    else: 
        return 2  # Major Delay 
 
# Example 
delays = [5, 30, 75, -10, 45] 
classes = [bucketize_delay(d) for d on delays] 
# Output: [0, 1, 2, 0, 1]
